# Gather and sort the incoming data

  &get_form_data;

# Set some server specific variables

  $header = "/usr/local/etc/httpd/cgi-bin/BBS/Template/Messages/Header/header_$MYDATA{'mid'}.dat";
  $body = "/usr/local/etc/httpd/cgi-bin/BBS/Template/Messages/Body/body_$MYDATA{'mid'}.txt";
  $reply_list = "/usr/local/etc/httpd/cgi-bin/BBS/Template/Messages/Replies/replies_$MYDATA{'mid'}.list";
  $page_name = "Template Message Board";
  $background = "/Images/Backgrounds/bkgnd.jpg";
  $generate_post_script = "http://www.nchgr.nih.gov/cgi-bin/BBS/Template/Scripts/generate_post.cgi?oldmid=$MYDATA{'mid'}";
  $line = "/Images/Icons/Lines/smalline.gif";
  $post_response_script = "post_response.cgi?oldmid=$MYDATA{'mid'}";
  $post_a_reply_button = "/Images/Buttons/post_a_reply.gif";
  $gen_first_page_script = "gen_first_page.cgi?mid=1&oldmid=1";
  $read_button = "/Images/Buttons/message_archive_top.gif";
  $get_last_update_script = "get_last_update.cgi";
  $last_modified_button = "/Images/Buttons/last_modified.gif";

# Find out who wrote this message

  open(HEADER,$header);
  while (<HEADER>)
    {
    @grepfields=split(/:/,$_);
    $lname = $grepfields[0];
    $fname = $grepfields[1];
    $email = $grepfields[2];
    $date = $grepfields[3];
    $subject = $grepfields[4];
    }
  close (HEADER);

# Gather the body of the message inputing a <P> for paragraph breaks

  $_ = `cat $body`;
  s/\n\n/<P>/g;
  $data = $_;

# Output the HTML version of the post!

  print "Content-type: text/html\n\n";
  print "<HTML><HEAD><TITLE>$page_name</TITLE></HEAD>";
  print "<BODY BACKGROUND=\"$background\">";
  print "<FONT SIZE=+1>";
  print "<B>Name:</B> $fname $lname<BR>";
  print "<B>Email:</B> <A HREF=\"mailto:$email\">$email</A><BR>";
  print "<B>Date:</B> $date<BR>";
  print "<B>Subject:</B> $subject<BR>";
  print "</FONT>";
  print "<B>Body:</B><BLOCKQUOTE>$data</BLOCKQUOTE>";
  print "<B>Replies:</B><BLOCKQUOTE>";

# Gather the replies from the reply list file

  open (REPLIES, $reply_list);
  while (<REPLIES>)
    {
    ($url,$subject1) = split(/:/, $_);
    print "<A HREF=\"$generate_post_script&mid=$url\">Subject$subject1</A><BR>";
    }
  close (REPLIES);    

# Continue outputing to HTML

  print "</BLOCKQUOTE>";
  print "<CENTER><P><IMG SRC=\"$line\"><P>";
  print "<A HREF=\"$post_response_script\">";
  print "<IMG SRC=\"$post_a_reply_button\" BORDER=0 ALT=\"[Post";
  print "a Message]\"></a>\n";
  print "<A HREF=\"$gen_first_page_script\">";
  print "<IMG SRC=\"$read_button\" BORDER=0 ALT=\"[Read";
  print "through Message Archive]\"></A>\n";
  print "<A HREF=\"$get_last_update_script\">";
  print "<IMG SRC=\"$last_modified_button\"";
  print "BORDER=0 ALT=\"[Last Modified List]\"></A>";
  print "</CENTER></BODY></HTML>";

#############################################################################

#############################################################################

sub get_form_data

  {

# This form parsing routine was adapted from a script by Robert John
# Mudry which was published by the Coriolis Group as freeware in "Serving
# The Web".  This is an excellent book which I recommend to any cgi
# programmer.

# First, split up $FORM_DATA into $NAME and $VALUE.  Store $VALUE
# in $MYDATA{$NAME} provided that $NAME is not already in use.  If it is, a
# temporary number stored in $NUM is incremented and a new the $VALUE is
# saved as the new variable.

  $FORM_DATA = $ENV{'QUERY_STRING'};
  foreach (split(/&/, $FORM_DATA))
    {
    ($NAME, $VALUE) = split(/=/, $_);
    $NAME =~ s/\+/ /g;
    $NAME =~ s/%([0-9|A-F]{2})/pack(C,hex($1))/eg;
    $VALUE =~ s/\+/ /g;
    $VALUE =~ s/%([0-9|A-F]{2})/pack(C,hex($1))/eg;

# Find unique Variable Name for every variable coming in from the form input

    $NUM = "0";
    while ($MYDATA{$NAME} ne "")
      {
      $NUM++;
      $NAME =~ s/\.([0-9]+$)|$/\.$NUM/;
      }
# Store Variable NAME=Variable VAUE pair

      $MYDATA{$NAME} = $VALUE;
    }
  }
